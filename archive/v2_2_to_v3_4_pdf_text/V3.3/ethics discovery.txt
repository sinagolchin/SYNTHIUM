@dataclass
class CareMetrics:
      """
      Quantifies "care" between two beings.
      Care = allocation of resources toward another's wellbeing.
      """
      resource_transfer: float          # How much capacity A gives to B
      action_alignment: float            # How much A's actions benefit B
      information_sharing: float         # How much A reduces B's uncertainty
      total_care: float                 # Composite measure


      @classmethod
      def measure(cls, being_A_state: RealityVector,
                           being_B_state: RealityVector,
                           being_A_action: Dict[str, float]) -> 'CareMetrics':
            """
            Measure care from A toward B.


            Args:
                  being_A_state: A's current state
                  being_B_state: B's current state
                  being_A_action: A's chosen action (changes to dimensions)
          Returns:
                CareMetrics quantifying A's care for B
          """
          # 1. Resource transfer (does A give capacity to B?)
          capacity_change = being_A_action.get('C', 0.0)
          resource_transfer = -capacity_change if capacity_change < 0 else 0.0


          # 2. Action alignment (does A's action increase B's enjoyment?)
          # Predict B's state after A's action affects field
          action_vector = np.array([being_A_action.get(d, 0) for d in ['v','R','r','C','S']])
          b_vector = np.array([being_B_state.v, being_B_state.R, being_B_state.r,
                                   being_B_state.C, being_B_state.S])


          # If A's action would move B toward lower entropy, higher resonance → care
          alignment = 0.0
          if being_A_action.get('S', 0) < 0:       # A reduces entropy
                alignment += 0.3
          if being_A_action.get('r', 0) > 0:       # A increases resonance
                alignment += 0.3
          if being_A_action.get('C', 0) > 0:       # A increases capacity
                alignment += 0.4


          # 3. Information sharing (does A reduce B's perception error?)
          # Measured by how much A's state reveals about optimal states
          info_sharing = being_A_state.P * 0.5       # Higher P = can share better info


          # Total care
          total = resource_transfer + alignment + info_sharing


          return cls(
                resource_transfer=resource_transfer,
                action_alignment=alignment,
                information_sharing=info_sharing,
                total_care=total
          )



@dataclass
class ConservationLaw:
    """
    A discovered conservation law in multi-agent systems.
    """
    name: str
    quantity: str                           # What's conserved (e.g., "total_E", "total_M")
    conserved: bool                         # Is it actually conserved?
    violation_tolerance: float              # How much variation is acceptable
    measurements: List[float] = field(default_factory=list)


    def check_conservation(self, new_measurement: float) -> bool:
          """Check if conservation still holds with new measurement."""
          self.measurements.append(new_measurement)


          if len(self.measurements) < 2:
                return True


          initial = self.measurements[0]
          variation = abs(new_measurement - initial) / (initial + 0.01)


          return variation < self.violation_tolerance



class EthicsDiscovery:

    def measure_care_network(self, beings: List[Tuple[RealityVector, Dict[str, float]]]) -> D
          """
          Measure care relationships in a network of beings.


          Args:
                beings: List of (state, action) tuples


          Returns:
                Care network matrix + analysis
          """
          n = len(beings)
          care_matrix = np.zeros((n, n))


          for i in range(n):
                for j in range(n):
                     if i == j:
                         continue


                     state_i, action_i = beings[i]
                     state_j, _ = beings[j]
               care = CareMetrics.measure(state_i, state_j, action_i)
               care_matrix[i, j] = care.total_care


    # Analyze network
    total_care = np.sum(care_matrix)
    reciprocal_care = np.sum(care_matrix * care_matrix.T) / 2       # Mutual care
    exploitative_care = np.sum(np.maximum(care_matrix - care_matrix.T, 0))          # One-way


    return {
          'care_matrix': care_matrix,
          'total_care': total_care,
          'reciprocal_care': reciprocal_care,
          'exploitative_care': exploitative_care,
          'network_balance': reciprocal_care / (total_care + 0.01)
    }

    if stable_configs:

          self.stable_patterns.extend(common_patterns)

    return []

    """

    If total E is conserved → zero-sum game (one's gain = another's loss)
    If total E increases → positive-sum possible
    If total M is conserved → meaning is transferred, not created
    """
    # Test various quantities
    candidates = {

          if len(measurements) < 2:

def _analyze_stability(self, history: Dict[str, List[RealityVector]]) -> Dict:
    """Analyze if system reached stable equilibrium."""
    # Check if final states are stable (not oscillating)
    stability_score = 0.0


    for name, trajectory in history.items():
        if len(trajectory) < 100:
            continue
        # Check last 100 steps for stability
        recent = trajectory[-100:]


        # Measure variance in each dimension
        variances = []
        for dim in ['v', 'R', 'r', 'C', 'S', 'E']:
               values = [getattr(s, dim) for s in recent]
               var = np.var(values)
               variances.append(var)


        # Low variance = stable
        avg_var = np.mean(variances)
        agent_stability = 1.0 / (1.0 + avg_var * 10)
        stability_score += agent_stability


    # Average across agents
    stability_score /= len(history)


    return {
        'is_stable': stability_score > 0.7,
        'score': stability_score
    }


def _extract_pattern(self, history: Dict[str, List[RealityVector]]) -> Dict:
    """Extract characteristic pattern from trajectory."""
    # Find average final state
    final_states = [traj[-1] for traj in history.values()]


    avg_pattern = {
        'v': np.mean([s.v for s in final_states]),
        'R': np.mean([s.R for s in final_states]),
        'r': np.mean([s.r for s in final_states]),
        'C': np.mean([s.C for s in final_states]),
        'S': np.mean([s.S for s in final_states]),
        'E': np.mean([s.E for s in final_states]),
    }


    return avg_pattern
