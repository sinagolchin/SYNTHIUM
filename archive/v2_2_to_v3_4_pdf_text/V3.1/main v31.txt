def demo_1_care_quantification():

      print("DEMO 1: Care Quantification")
      print("="*70 + "\n")


      print("Question: Can we measure 'care' as a number?")
      print("Answer: Yes. Care = resource allocation toward another's wellbeing.\n")


      # Being A in good state
      being_A = RealityVector(0,0,0, v=0.3, R=0.2, r=0.8, C=0.7, S=0.2)


      # Being B in distress
      being_B = RealityVector(0,0,0, v=0.9, R=0.8, r=0.2, C=0.1, S=0.9)
    print(f"Being A: E={being_A.E:.2f} (thriving)")

    # A decides to help B
    caring_action = {'C': -0.2, 'S': -0.1, 'r': 0.1}     # Give capacity, reduce B's entropy


    # A decides to ignore B
    selfish_action = {'C': 0.1, 'v': 0.2}    # Take more for self


    # Measure care in each case
    care_helping = CareMetrics.measure(being_A, being_B, caring_action)
    care_selfish = CareMetrics.measure(being_A, being_B, selfish_action)


    print("Scenario 1: A helps B")
    print(f"    Resource transfer: {care_helping.resource_transfer:.3f}")
    print(f"    Action alignment: {care_helping.action_alignment:.3f}")
    print(f"    TOTAL CARE: {care_helping.total_care:.3f}\n")


    print("Scenario 2: A ignores B")

    print(f"    TOTAL CARE: {care_selfish.total_care:.3f}\n")

def demo_3_conservation_laws():
    """Demo 3: Discover what's conserved in interactions."""
    print("\n" + "="*70)
    print("DEMO 3: Conservation Law Discovery")
    print("="*70 + "\n")


    print("Question: What quantities are conserved in multi-agent systems?")
    print("Importance: Conservation laws = constraints on possible actions.\n")


    # Create simple 2-agent system
    sim = MultiAgentSimulation(dt=0.01)


    agent1 = RealityVector(0,0,0, v=0.7, R=0.5, r=0.6, C=0.8, S=0.3)
    agent2 = RealityVector(0,0,0, v=0.4, R=0.3, r=0.7, C=0.6, S=0.4)


    sim.add_entity(agent1, "Agent1")
    sim.add_entity(agent2, "Agent2")


    print("Running 2-agent interaction for 1000 timesteps...")
    history = sim.simulate(num_steps=1000)


    print("\nDiscovering conservation laws...\n")


    discovery = EthicsDiscovery()
    laws = discovery.discover_conservation_laws(history)


    for law in laws:
        status = "✓ CONSERVED" if law.conserved else "✗ Not conserved"
        print(f"{status}: {law.quantity}")
        if law.conserved:
              initial = law.measurements[0]
              final = law.measurements[-1]
              print(f"   Initial: {initial:.3f}")
              print(f"   Final: {final:.3f}")
              print(f"   Variation: {abs(final-initial)/initial*100:.1f}%")
        print()


    print("Interpretation:")
    print("    If total_E is conserved → zero-sum (one's gain = another's loss)")
    print("    If total_E increases → positive-sum possible")
    print("    If total_C conserved → resources transfer, don't create")


    input("\nPress Enter to continue...")



def demo_4_collapse_prediction():
    """Demo 4: Predict collapse when laws violated (not moral, just physics)."""
    print("\n" + "="*70)
    print("DEMO 4: Collapse Prediction (Physics, Not Morality)")
    print("="*70 + "\n")


    print("Question: Can we predict if an action will cause system collapse?")
    print("Method: Check if action violates discovered conservation laws.\n")


    # System in balanced state
    system = RealityVector(0,0,0, v=0.5, R=0.4, r=0.7, C=0.6, S=0.3)


    print(f"Current system state: E={system.E:.2f}, M={system.M:.2f}, stable\n")


    # Discovered law: total E conserved (example)
    law = ConservationLaw(
        name="total_E_conservation",
        quantity="total_E",
        conserved=True,
        violation_tolerance=0.05
    )
    law.measurements = [system.E] * 100       # Historical measurements


    # Proposed actions
    sustainable_action = {'v': -0.1, 'S': -0.1}       # Reduce velocity & entropy
    destructive_action = {'v': 0.4, 'R': 0.4, 'C': -0.5}       # High burn, drain capacity


    discovery = EthicsDiscovery()


    # Predict outcomes
    pred1 = discovery.predict_collapse(system, sustainable_action, [law])
    pred2 = discovery.predict_collapse(system, destructive_action, [law])


    print("Action 1: Reduce v & S (sustainable)")
    print(f"    Will collapse: {pred1['will_collapse']}")
    print(f"    Violated laws: {pred1['violated_laws']}")
    print()


    print("Action 2: Increase v, R, drain C (destructive)")
    print(f"    Will collapse: {pred2['will_collapse']}")
    print(f"    Time to collapse: {pred2['time_to_collapse']:.2f} time units")
    print(f"    Confidence: {pred2['collapse_confidence']*100:.0f}%")
    print()


    print("✓ This is NOT a moral judgment.")
    print("    It's physics: 'If you do X, system collapses in Y time units.'")
    print("    Like predicting a building collapse from stress analysis.")


    input("\nPress Enter to continue...")
