import numpy as np

class HolisticScience:
      """Unified physics engine for 11D reality space."""


      def __init__(self):
          self.field = RealityField()


      def analyze_stability(self, vector: RealityVector) -> Dict[str, Any]:
          """Predict stability, burn rate, time-to-failure."""


          # Special case: awakening fixed point
          if vector.v < 0.01 and vector.r > 0.99 and vector.S < 0.01:
              return {
                  "coherence_status": "Stable Fixed Point (Awakening)",
                  "note": "v=0, r=1.0, S=0 is a stable equilibrium",
                  "burn_rate": 0.0,
                  "predicted_lifespan": "Infinite",
                  "risk_factors": []
              }


          burn_rate = vector.v * vector.R
          fuel = vector.C
          integrity = 1.0 - vector.S
          sustainability_limit = 0.4 * integrity


          analysis = {
              "coherence_status": "Stable",
              "burn_rate": f"{burn_rate:.3f} units/t",
              "predicted_lifespan": "Indefinite",
              "risk_factors": []
          }
    # Hollow titan check
    if vector.v > 0.8 and vector.C < 0.3:
        analysis["risk_factors"].append("HOLLOW STRUCTURE: High speed without depth")
        analysis["coherence_status"] = "Fragile"


    # Burnout calculation
    if burn_rate > sustainability_limit:
        analysis["coherence_status"] = "Meta-Stable (Decaying)"
        excess_burn = burn_rate - sustainability_limit
        ttf = fuel / (excess_burn + 0.001)
        analysis["predicted_lifespan"] = f"{ttf:.2f} Time Units"


        if ttf < 2.0:
            analysis["coherence_status"] = "CRITICAL: IMMINENT COLLAPSE"
            analysis["risk_factors"].append("System crash imminent")


    # Intervention options
    analysis["intervention_options"] = [
        f"Reduce velocity v = {vector.v:.2f} → {vector.v*0.5:.2f}",
        f"Reduce resistance R = {vector.R:.2f} → {vector.R*0.5:.2f}",
        f"Increase capacity C = {vector.C:.2f} → {min(1.0, vector.C+0.3):.2f}",
        f"Reduce entropy S = {vector.S:.2f} → {vector.S*0.5:.2f}"
    ]


    # Predict equilibrium
    equilibrium = self.field.find_equilibrium(vector, max_steps=100)
    analysis["predicted_equilibrium"] = {
        "will_converge": True,
        "equilibrium_state": equilibrium.to_dict(),
        "equilibrium_enjoyment": equilibrium.E,
        "enjoyment_change": equilibrium.E - vector.E
    }


    return analysis


def predict_trajectory(self, vector: RealityVector,
                        time_horizon: int = 100) -> Dict[str, Any]:
    """Predict evolution over time_horizon timesteps."""
    trajectory = self.field.evolve(vector, num_steps=time_horizon)
    final = trajectory[-1]


    initial_pot = self.field.compute_potential(vector)
    final_pot = self.field.compute_potential(final)


    changes = {}
    for dim in ['v','R','r','C','S','E','M','P']:
        ival = getattr(vector, dim)
        fval = getattr(final, dim)
        changes[dim] = {
               'initial': ival,
               'final': fval,
               'delta': fval - ival
        }


    return {
        'trajectory': {
               'length': len(trajectory),
               'initial_state': vector.to_dict(),
               'final_state': final.to_dict(),
               'converged_to_equilibrium': final_pot < initial_pot
        },
        'changes': changes,
        'potential': {
               'initial': initial_pot,
               'final': final_pot,
               'decreased': final_pot < initial_pot
        }
    }


def compare_states(self, state_a: RealityVector,
                      state_b: RealityVector) -> Dict[str, Any]:
    """Compare two reality states."""
    distance = state_a.distance_to(state_b)
    comparison = {"distance_8D": distance, "dimension_differences": {}}


    for dim in ['v','R','r','C','S','E','M','P']:
        val_a = getattr(state_a, dim)
        val_b = getattr(state_b, dim)
        comparison["dimension_differences"][dim] = {
               'state_a': val_a,
               'state_b': val_b,
               'delta': val_b - val_a
        }


    comparison["higher_enjoyment"] = "state_a" if state_a.E > state_b.E else \
                                         "state_b" if state_b.E > state_a.E else "equal"
    return comparison


def analyze_field(self, vectors: List[RealityVector]) -> Dict[str, Any]:
    """Statistics over collection of states."""
    if not vectors:
        return {"error": "No vectors provided"}


    stats = {"num_states": len(vectors), "statistics": {}}
for dim in ['v','R','r','C','S','E','M','P']:
    vals = [getattr(v, dim) for v in vectors]
    stats["statistics"][dim] = {
        'mean': float(np.mean(vals)),
        'std': float(np.std(vals)),
        'min': float(np.min(vals)),
        'max': float(np.max(vals))
    }


avg_E = np.mean([v.E for v in vectors])
avg_S = np.mean([v.S for v in vectors])


if avg_S > 0.7:
    stats["field_character"] = "High-entropy chaotic field"
elif avg_E > 0.7:
    stats["field_character"] = "High-enjoyment flourishing field"
else:
    stats["field_character"] = "Mixed field"


return stats
