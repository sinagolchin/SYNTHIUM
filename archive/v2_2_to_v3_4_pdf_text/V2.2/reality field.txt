from datetime import datetime
from typing import List, Dict, Any
from .vectors_v2 import RealityVector



class RealityField:
      """Manages collections of reality vectors."""


      def __init__(self, name: str):
          self.name = name
          self.vectors: List[RealityVector] = []
          self.metadata = {
              "created": datetime.now().isoformat(),
              "framework_version": "2.2"
          }


      def add_vector(self, vector: RealityVector):
          """Add a vector to this field."""
          self.vectors.append(vector)


      def to_dict(self) -> Dict[str, Any]:
          """Serialize field to dictionary."""
          return {
              "name": self.name,
              "metadata": self.metadata,
              "vector_count": len(self.vectors),
              "vectors": [vec.to_dict() for vec in self.vectors]
          }


      @classmethod
      def from_dict(cls, data: Dict[str, Any]) -> 'RealityField':
          """Deserialize field from dictionary."""
          field = cls(data['name'])
          field.metadata = data['metadata']
          field.vectors = [RealityVector.from_dict(vec) for vec in data['vectors']]
          return field


      def __len__(self) -> int:
          return len(self.vectors)
def __str__(self) -> str:
    return f"RealityField('{self.name}', {len(self.vectors)} vectors)"
