from dataclasses import dataclass, field

@dataclass
class Glyph:
      """
      An archetypal pattern or experiential quality.
      """
      name: str
      domains: Dict[str, float]
      description: str = ""



@dataclass
class Soil:
      """
      The Living Soil - collection of archetypal glyphs.
      """
      glyphs: Dict[str, Glyph] = field(default_factory=dict)


      def plant(self, glyph: Glyph):
            """Add a glyph to the soil."""
            self.glyphs[glyph.name] = glyph

@staticmethod
def transduce_glyph(glyph: Glyph) -> RealityVector:
    """
    Convert a Glyph to an 8D RealityVector.
    """
    # SPECIAL CASE: Awakening (Sakshatkara)
    awakening_glyphs = ["I Am", "This", "Silence", "Tat Tvam Asi"]
    if glyph.name in awakening_glyphs:
          return RealityVector(
               x=0.0, y=0.0, z=0.0,
               v=0.0,   # No movement
               R=0.0,   # No resistance
               r=1.0,   # Perfect resonance
               C=1.0,   # Infinite capacity
               S=0.0    # Zero entropy
          )


    # Standard Mapping
    vals = {'v': 0.1, 'R': 0.1, 'r': 0.1, 'C': 0.1, 'S': 0.1}


    d = glyph.domains


    if "Action" in d:
    vals['v'] = d["Action"]


if "Chaos" in d:
    vals['S'] = d["Chaos"]


if "Love" in d or "Spiritual" in d:
    vals['r'] = max(d.get("Love", 0), d.get("Spiritual", 0))


if "Mystical" in d and d["Mystical"] < 1.0:
    vals['r'] = d["Mystical"]


if "Symbolic" in d or "Memory" in d:
    vals['C'] = max(d.get("Symbolic", 0), d.get("Memory", 0))


if "Action" in d and d["Action"] > 0.7:
    vals['R'] = min(d["Action"] * 0.6, 1.0)


# Ensure all values in [0, 1]
for key in vals:
    vals[key] = max(0.0, min(1.0, vals[key]))


return RealityVector(x=0.0, y=0.0, z=0.0, **vals)
