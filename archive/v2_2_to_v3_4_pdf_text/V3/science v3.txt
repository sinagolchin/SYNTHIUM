    "perception_accuracy": vector.P

# Check for awakening state (v=0, r=1.0, S=0)
if abs(vector.v) < 0.01 and vector.r > 0.99 and vector.S < 0.01:

# Predict equilibrium

def predict_trajectory(self,
                         vector: RealityVector,
                         time_horizon: int = 100) -> Dict[str, Any]:
    """
    Predict how system will evolve over time.


    Args:
          vector: Initial state
          time_horizon: Number of timesteps to predict


    Returns:
          Trajectory analysis
    """
    trajectory = self.field.evolve(vector, num_steps=time_horizon)
    analysis = self.field.analyze_trajectory(trajectory)


    # Add predictions
    final = trajectory[-1]


    analysis["trajectory"] = {
          "length": len(trajectory),
          "initial_state": vector.to_dict(),
          "final_state": final.to_dict(),
          "converged_to_equilibrium": analysis['potential']['decreased']
    }


    return analysis


def compare_states(self,
                     state_a: RealityVector,
                     state_b: RealityVector) -> Dict[str, Any]:
    """
    Compare two reality states.
    """
    distance = state_a.distance_to(state_b)


    comparison = {
          "distance_8D": distance,
          "dimension_differences": {}
    }


    for dim in ['v', 'R', 'r', 'C', 'S', 'E', 'M', 'P']:
          val_a = getattr(state_a, dim)
          val_b = getattr(state_b, dim)


          comparison["dimension_differences"][dim] = {
              'state_a': val_a,
              'state_b': val_b,
              'delta': val_b - val_a,
              'percent_diff': ((val_b - val_a) / (val_a + 0.01)) * 100
          }


    # Which state is "better"?
    if state_a.E > state_b.E:
          comparison["higher_enjoyment"] = "state_a"
    elif state_b.E > state_a.E:
          comparison["higher_enjoyment"] = "state_b"
    else:
          comparison["higher_enjoyment"] = "equal"


    return comparison


def analyze_field(self, vectors: List[RealityVector]) -> Dict[str, Any]:
    """
    Analyze a field of multiple states.
    """
    if not vectors:
          return {"error": "No vectors provided"}


    analysis = {
          "num_states": len(vectors),
          "statistics": {}
    }


    # Compute statistics for each dimension
    for dim in ['v', 'R', 'r', 'C', 'S', 'E', 'M', 'P']:
          values = [getattr(v, dim) for v in vectors]


          analysis["statistics"][dim] = {
              'mean': np.mean(values),
              'std': np.std(values),
              'min': np.min(values),
              'max': np.max(values),
              'range': np.max(values) - np.min(values)
          }
    # Overall field characteristics
    avg_enjoyment = np.mean([v.E for v in vectors])
    avg_entropy = np.mean([v.S for v in vectors])


    if avg_entropy > 0.7:
          analysis["field_character"] = "High-entropy chaotic field"
    elif avg_enjoyment > 0.7:
          analysis["field_character"] = "High-enjoyment flourishing field"
    elif avg_entropy < 0.3 and avg_enjoyment > 0.5:
          analysis["field_character"] = "Ordered and thriving field"
    else:
          analysis["field_character"] = "Mixed field"


    return analysis


def find_optimal_state(self,
                         current: RealityVector,
                         constraints: Dict[str, tuple] = None) -> RealityVector:
    """
    Find optimal state from current position.


    Args:
          current: Starting state
          constraints: Dict of {dimension: (min, max)}


    Returns:
          Optimized state (maximum enjoyment within constraints)
    """
    # Start from equilibrium
    optimal = self.field.find_equilibrium(current, max_steps=500)


    # Apply constraints if provided
    if constraints:
          for dim, (min_val, max_val) in constraints.items():
               if hasattr(optimal, dim):
                   current_val = getattr(optimal, dim)
                   constrained = np.clip(current_val, min_val, max_val)
                   setattr(optimal, dim, constrained)


          # Recompute derived
          optimal.E = optimal.compute_enjoyment()
          optimal.M = optimal.compute_meaning()
          optimal.P = optimal.compute_perception_accuracy()


    return optimal
