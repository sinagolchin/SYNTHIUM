from dataclasses import dataclass
from typing import List, Optional, Dict
from enum import Enum



class ConfidenceLevel(Enum):
      """Standardized confidence levels."""
      HIGH = "HIGH"
      MODERATE = "MODERATE"
      LOW = "LOW"
      SPECULATIVE = "SPECULATIVE"
      UNKNOWN = "UNKNOWN"



@dataclass
class Confidence:
      """
      Explicit confidence tracking for any claim.


      Every measurement, formula, and prediction gets a Confidence object.
      """
      level: ConfidenceLevel
      basis: str    # Why we have this confidence level
      limitations: List[str]       # Known limitations
      unknowns: List[str]     # Known unknowns (things we know we don't know)


      def __repr__(self):
            return f"{self.level.value}: {self.basis}"


      def full_report(self) -> str:
            """Generate full confidence report."""
            lines = [
                f"Confidence: {self.level.value}",
                f"Basis: {self.basis}"
            ]


            if self.limitations:
              lines.append("Known Limitations:")
              for lim in self.limitations:
                  lines.append(f"   - {lim}")


         if self.unknowns:
              lines.append("Known Unknowns:")
              for unk in self.unknowns:
                  lines.append(f"   - {unk}")


         return "\n".join(lines)

# MEASUREMENT CONFIDENCE HELPERS
# ==================================================


def create_measurement_confidence(
    value: float,
    measurement_type: str,
    source: str,
    reliability: ConfidenceLevel
) -> Confidence:
    """
    Helper to create confidence for a measurement.


    Args:
          value: The measured value
          measurement_type: What was measured (e.g., 'violence', 'capacity')
          source: Source of measurement
          reliability: Assessed reliability level


    Returns:
          Confidence object
    """


    # Common limitations by measurement type
    type_limitations = {
          'violence': ['Casualty counts may be incomplete', 'Measurement during chaos'],
          'capacity': ['Economic data may be outdated', 'Black market not captured'],
          'resonance': ['Polling may have bias', 'Fear affects responses'],
          'entropy': ['Subjective assessment', 'May miss hidden order'],
          'resistance': ['Leadership opacity', 'Internal factions unclear']
    }


    # Common unknowns
    type_unknowns = {
          'violence': ['Hidden casualties', 'Unreported incidents'],
          'capacity': ['Underground economy', 'External support'],
          'resonance': ['Silent majority', 'Coerced compliance'],
          'entropy': ['Emergent order', 'Hidden coordination'],
          'resistance': ['Internal power struggles', 'Succession plans']
    }


    return Confidence(
          level=reliability,
          basis=f'{measurement_type}={value:.2f} from {source}',
          limitations=type_limitations.get(measurement_type, ['Measurement uncertainty']),
          unknowns=type_unknowns.get(measurement_type, ['Unknown factors'])
    )

# UNCERTAINTY AGGREGATION
# ==================================================


def aggregate_confidence(confidences: List[Confidence]) -> Confidence:
    """
    Aggregate multiple confidence levels.


    Rule: Weakest link principle - overall confidence is lowest component.
    """
    levels = [c.level for c in confidences]


    # Find weakest confidence
    if ConfidenceLevel.SPECULATIVE in levels:
          weakest = ConfidenceLevel.SPECULATIVE
    elif ConfidenceLevel.LOW in levels:
          weakest = ConfidenceLevel.LOW
    elif ConfidenceLevel.MODERATE in levels:
          weakest = ConfidenceLevel.MODERATE
    else:
          weakest = ConfidenceLevel.HIGH


    # Aggregate limitations and unknowns
    all_limitations = []
    all_unknowns = []
    for c in confidences:
          all_limitations.extend(c.limitations)
          all_unknowns.extend(c.unknowns)


    return Confidence(
          level=weakest,
          basis=f'Aggregated from {len(confidences)} components',
          limitations=list(set(all_limitations)),   # Deduplicate
          unknowns=list(set(all_unknowns))
    )
