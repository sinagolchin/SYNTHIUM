    pessimistic: float
    optimistic: float


    # Confidence breakdown
    data_confidence: Confidence
    model_confidence: Confidence
    measurement_confidence: Confidence
    overall_confidence: Confidence

    key_factors: List[str]

    agnosticism_note: str



class AgnosticEmpiricalDiscovery:
    """
    Empirical discovery with EXPLICIT UNCERTAINTY.


    All predictions come with:
    - Confidence levels
    - Uncertainty ranges
    - Known limitations
    - Sample size warnings
    - Framework uncertainty disclaimers
    """


    def __init__(self, cases: List[InterventionCase] = None):
          self.cases = cases if cases else HISTORICAL_CASES
          self.discovered_patterns = None


    def discover_stability_predictors(self) -> Dict[str, float]:
          """Same as v3.3 - discover correlations."""
          if len(self.cases) < 3:
              return {"error": "Need at least 3 cases"}


          features = []
          outcomes = []


          for case in self.cases:
              init = case.initial_state
              interv = case.intervener_state


              feature = {
                  'initial_chaos': init.S,
                  'initial_violence': init.v,
                  'local_support': case.local_support,
                  'force_ratio': case.military_force,
                  'intervener_capacity': interv.C,
                  'resonance_gap': abs(init.r - interv.r),
                  'initial_capacity': init.C,
                  'duration': case.duration_years
              }
          features.append(feature)
          outcomes.append(case.stability_score)


    correlations = {}
    for key in features[0].keys():
          values = [f[key] for f in features]
          corr = np.corrcoef(values, outcomes)[0, 1]
          correlations[key] = corr


    self.discovered_patterns = correlations
    return correlations

def _find_similar_case(self, features: Dict[str, float]) -> Optional[InterventionCase]:
    """Find most similar historical case (same as v3.3)."""
    if not self.cases:
        return None


    min_distance = float('inf')
    most_similar = None


    for case in self.cases:
        init = case.initial_state
        interv = case.intervener_state


        case_features = {
            'initial_chaos': init.S,
            'initial_violence': init.v,
            'local_support': case.local_support,
            'force_ratio': case.military_force,
            'intervener_capacity': interv.C,
            'resonance_gap': abs(init.r - interv.r),
            'initial_capacity': init.C,
            'duration': case.duration_years
        }


        distance = sum((features[k] - case_features[k])**2 for k in features.keys())


        if distance < min_distance:
            min_distance = distance
            most_similar = case


    return most_similar


def _compute_similarity_score(self, features: Dict, similar_case: InterventionCase) -> fl
    """Compute how similar this case is to the most similar historical case."""
    if similar_case is None:
        return 0.0


    init = similar_case.initial_state
    interv = similar_case.intervener_state


    case_features = {
        'initial_chaos': init.S,
        'initial_violence': init.v,
        'local_support': similar_case.local_support,
        'force_ratio': similar_case.military_force,
        'intervener_capacity': interv.C,
        'resonance_gap': abs(init.r - interv.r),
        'initial_capacity': init.C,
        'duration': similar_case.duration_years
    }


    distance = np.sqrt(sum((features[k] - case_features[k])**2 for k in features.keys()))
    similarity = 1.0 / (1.0 + distance)


    return similarity
